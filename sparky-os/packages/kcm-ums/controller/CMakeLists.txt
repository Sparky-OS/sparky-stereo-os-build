add_library(umsrenderer STATIC backend.cpp)
target_include_directories(umsrenderer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../src)
target_link_libraries(umsrenderer PUBLIC umsprofile Qt6::Network Qt6::Xml KF6::I18n)
add_executable(ums-renderer main.cpp window.cpp mpris.cpp)
target_link_libraries(ums-renderer PRIVATE umsrenderer Qt6::Widgets Qt6::DBus)
install(TARGETS ums-renderer DESTINATION ${KDE_INSTALL_BINDIR})
add_subdirectory(purpose)
if(BUILD_TESTING)
    find_package(KF6KIO REQUIRED)
    add_executable(umsrenderertest tests/renderertest.cpp window.cpp mpris.cpp)
    target_link_libraries(umsrenderertest PRIVATE umsrenderer Qt6::Widgets Qt6::DBus Qt6::Test KF6::Purpose KF6::KIOWidgets)
    target_compile_definitions(umsrenderertest PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
endif()
install(FILES ums-renderer.desktop DESTINATION ${KDE_INSTALL_APPDIR})
install(FILES ums-play.desktop DESTINATION ${KDE_INSTALL_DATADIR}/kio/servicemenus)
if(BUILD_TESTING)
    add_executable(umsrealservertest tests/real-server.cpp)
    target_link_libraries(umsrealservertest PRIVATE umsrenderer)
endif()
install(FILES ums-renderer.1 DESTINATION ${KDE_INSTALL_MANDIR}/man1)
