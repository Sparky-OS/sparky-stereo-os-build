cmake_minimum_required(VERSION 3.22)
project(sparky-stereo-cec VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
find_package(KF6Config REQUIRED)
find_package(KF6Screen REQUIRED)
add_library(cec-adapter STATIC src/adapter.cpp src/adapter.h src/discovery.cpp)
target_include_directories(cec-adapter PUBLIC src)
target_link_libraries(cec-adapter PUBLIC Qt6::Core)
add_library(cec-controller STATIC src/controller.cpp src/controller.h)
target_link_libraries(cec-controller PUBLIC cec-adapter Qt6::Gui KF6::ScreenDpms)
add_executable(sparky-stereo-cec src/main.cpp)
target_link_libraries(sparky-stereo-cec PRIVATE cec-controller KF6::ConfigCore)
install(TARGETS sparky-stereo-cec DESTINATION bin)
option(BUILD_TESTING "Build CEC tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()
