cmake_minimum_required(VERSION 3.16)
project(media-gpu-route VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
find_package(Vulkan REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(VAAPI REQUIRED IMPORTED_TARGET libva libva-drm)
pkg_check_modules(NVCODEC REQUIRED IMPORTED_TARGET ffnvcodec)
add_library(media-gpu-route SHARED src/media-gpu-route.c src/vaapi-caps.c src/cuda-caps.c)
target_include_directories(media-gpu-route PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(media-gpu-route PRIVATE Vulkan::Vulkan PkgConfig::DRM PkgConfig::VAAPI PkgConfig::NVCODEC ${CMAKE_DL_LIBS})
target_compile_options(media-gpu-route PRIVATE -Wall -Wextra -Werror)
set_target_properties(media-gpu-route PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
add_executable(media-gpu-route-query src/main.c)
target_link_libraries(media-gpu-route-query PRIVATE media-gpu-route)
configure_file(media-gpu-route.pc.in media-gpu-route.pc @ONLY)
install(TARGETS media-gpu-route media-gpu-route-query LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/media-gpu-route.h include/media-gpu-route-ffmpeg.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/media-gpu-route.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
include(CTest)
if(BUILD_TESTING)
    add_executable(profile-caps-test tests/profile-caps.c)
    target_include_directories(profile-caps-test PRIVATE include src)
    target_link_libraries(profile-caps-test PRIVATE media-gpu-route)
    target_compile_options(profile-caps-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME profile-caps COMMAND profile-caps-test)
endif()
