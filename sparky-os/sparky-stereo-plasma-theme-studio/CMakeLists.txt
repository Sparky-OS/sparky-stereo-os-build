cmake_minimum_required(VERSION 3.20)
project(sparky-stereo-plasma-theme-studio VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(identity_output "${CMAKE_CURRENT_BINARY_DIR}/identity")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/identity.py" "${CMAKE_CURRENT_SOURCE_DIR}" "${identity_output}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS identity.json identity.py)
include_directories("${identity_output}")
file(GLOB authored_catalogues CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lang/*.ts")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${authored_catalogues})
include(GNUInstallDirs)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Xml Svg DBus LinguistTools)
find_package(KF6Config REQUIRED)
option(STUDIO_STEREO "Build the optional stereo authoring layer" ON)
if(STUDIO_STEREO)
    find_package(StereoLayer 1.18 REQUIRED)
    set(depth_backend stereo/depth.cpp)
    set(preview_backend stereo/preview.cpp)
    set(stereo_export stereo/export.cpp)
else()
    set(depth_backend plain/depth.cpp)
    set(preview_backend plain/preview.cpp)
endif()
add_compile_definitions(STUDIO_STEREO=$<BOOL:${STUDIO_STEREO}>)
set(depth_input "${CMAKE_CURRENT_SOURCE_DIR}/geometry/kwin")
set(depth_output "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(STUDIO_STEREO)
add_custom_command(OUTPUT "${depth_output}/desktop-depth.h" "${depth_output}/depth-source.json"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/extract-depth.py" "${depth_input}" "${depth_output}"
    DEPENDS extract-depth.py geometry/kwin/src/scene/stereodepth.cpp geometry/kwin/src/effect/globals.h geometry/kwin/src/kwin.kcfg
    VERBATIM)
endif()
add_library(theme-drawing STATIC drawing.cpp drawing.h ${depth_backend} ${stereo_export} depth-interface.h)
target_include_directories(theme-drawing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${depth_output}")
target_link_libraries(theme-drawing PUBLIC Qt6::Widgets Qt6::Xml Qt6::Svg KF6::ConfigCore)
add_library(theme-window STATIC editor.cpp parts.cpp preview.cpp ${preview_backend} editor.h preview.h)
target_link_libraries(theme-window PUBLIC theme-drawing Qt6::DBus)
if(STUDIO_STEREO)
    target_sources(theme-drawing PRIVATE "${depth_output}/desktop-depth.h")
    target_link_libraries(theme-window PUBLIC Stereo::Layer)
endif()
option(STUDIO_MODELS "Build the model authoring helper" ${STUDIO_STEREO})
if(STUDIO_MODELS AND NOT STUDIO_STEREO)
    message(FATAL_ERROR "Model stereo authoring requires STUDIO_STEREO; disable STUDIO_MODELS for the ordinary core")
endif()
if(STUDIO_MODELS)
    find_package(VTK 9.6 REQUIRED COMPONENTS CommonCore RenderingCore RenderingOpenGL2 IOCore IOImport IOImage)
    target_sources(theme-window PRIVATE models/page.cpp models/assets.cpp)
    add_executable(studio-model-renderer models/renderer.cpp)
    target_link_libraries(studio-model-renderer PRIVATE theme-drawing ${VTK_LIBRARIES})
    vtk_module_autoinit(TARGETS studio-model-renderer MODULES ${VTK_LIBRARIES})
    install(TARGETS studio-model-renderer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_compile_definitions(STUDIO_MODELS=$<BOOL:${STUDIO_MODELS}>)
add_executable(sparky-stereo-plasma-theme-studio main.cpp)
target_link_libraries(sparky-stereo-plasma-theme-studio PRIVATE theme-window)
file(GLOB catalogues CONFIGURE_DEPENDS "${identity_output}/lang/*.ts")
set(compiled_catalogues)
foreach(catalogue IN LISTS catalogues)
    get_filename_component(name "${catalogue}" NAME_WE)
    set(qm "${CMAKE_CURRENT_BINARY_DIR}/translations/${name}.qm")
    add_custom_command(OUTPUT "${qm}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/translations"
        COMMAND Qt6::lrelease -nounfinished "${catalogue}" -qm "${qm}"
        DEPENDS "${catalogue}" VERBATIM)
    list(APPEND compiled_catalogues "${qm}")
endforeach()
add_custom_target(translations ALL DEPENDS ${compiled_catalogues})
install(TARGETS sparky-stereo-plasma-theme-studio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${compiled_catalogues} DESTINATION "${CMAKE_INSTALL_DATADIR}/sparky-stereo-plasma-theme-studio/translations")
install(FILES ${identity_output}/sparky-stereo-plasma-theme-studio.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES ${identity_output}/sparky-stereo-plasma-theme-studio.1 DESTINATION "${CMAKE_INSTALL_MANDIR}/man1")
if(BUILD_TESTING AND STUDIO_STEREO)
    add_executable(compensation-proof compensation-proof.cpp)
    target_link_libraries(compensation-proof PRIVATE theme-drawing)
    add_test(NAME compensated-pairs COMMAND compensation-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/project/icons/scalable/devices/video-television.svgz" "${CMAKE_CURRENT_BINARY_DIR}/compensated")
    add_test(NAME measured-compensation COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/measure-compensation.py" "${CMAKE_CURRENT_BINARY_DIR}/compensated")
    set_tests_properties(measured-compensation PROPERTIES DEPENDS compensated-pairs)
    add_executable(project-proof project-proof.cpp)
    target_link_libraries(project-proof PRIVATE theme-window)
    add_test(NAME shared-project COMMAND project-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata")
    add_executable(ui-proof ui-proof.cpp)
    target_link_libraries(ui-proof PRIVATE theme-window)
    add_executable(catalogue-proof catalogue-proof.cpp)
    target_link_libraries(catalogue-proof PRIVATE theme-window)
    add_executable(native-proof native-proof.cpp)
    target_link_libraries(native-proof PRIVATE theme-drawing)
    add_executable(editor-control editor-control.cpp)
    target_link_libraries(editor-control PRIVATE Qt6::Core Qt6::DBus)
    add_test(NAME controls COMMAND ui-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/project")
    add_test(NAME catalogues COMMAND catalogue-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/project" "${CMAKE_CURRENT_SOURCE_DIR}/lang" "${CMAKE_CURRENT_BINARY_DIR}/translations")
    add_test(NAME native-pairs COMMAND native-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/project/icons/scalable/devices/video-television.svgz" "${CMAKE_CURRENT_BINARY_DIR}/native-proof-views")
    add_test(NAME png-fields-disparity COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/measure.py" "${CMAKE_CURRENT_BINARY_DIR}/native-proof-views")
    set_tests_properties(png-fields-disparity PROPERTIES DEPENDS native-pairs)
    set_tests_properties(controls catalogues native-pairs shared-project compensated-pairs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LC_ALL=C.UTF-8;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()

if(BUILD_TESTING AND STUDIO_MODELS)
    add_executable(model-proof models/proof.cpp)
    target_link_libraries(model-proof PRIVATE theme-window)
    add_test(NAME models COMMAND model-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/models/fixture.gltf" "${CMAKE_CURRENT_BINARY_DIR}/model-proof-output")
    set_tests_properties(models PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LIBGL_ALWAYS_SOFTWARE=1;LC_ALL=C.UTF-8;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()

    add_test(NAME model-glb COMMAND model-proof "${CMAKE_CURRENT_SOURCE_DIR}/testdata/models/fixture.glb" "${CMAKE_CURRENT_BINARY_DIR}/model-glb-proof-output")
